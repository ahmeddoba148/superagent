#!/usr/bin/env bash
set -euo pipefail
echo "SAND PRIME gate start"
echo "===== Assemble exact PRIME source and static gates ====="
set -euo pipefail
cat sand_prime/part00.b64 sand_prime/part01.b64 sand_prime/part02.b64 sand_prime/part03.b64 | base64 -d | gzip -d > SAND.js
OLD=$(sha256sum SAND.js | awk '{print $1}')
echo "base_source_sha=$OLD"
test "$OLD" = "16225f81336429d1859ae9c4565d97578572157cf910b45629c03aa6ce18da1f"
patch --batch SAND.js < sand_prime/latest.patch
SHA=$(sha256sum SAND.js | awk '{print $1}')
echo "prime_source_sha=$SHA"
test "$SHA" = "8f41ce84e6c07104907041d297ce27c6ac143361d6211a38db05e394c56897ea"
node --check SAND.js
! grep -q 'text\.includes(' SAND.js
! grep -q 'Math\.random' SAND.js
! grep -q '@ts-ignore\|@ts-nocheck' SAND.js
! grep -Eq '\beval\(' SAND.js
npm install --no-save typescript@5.8.3 wrangler@latest >/dev/null
printf '%s' '{"compilerOptions":{"allowJs":true,"checkJs":true,"noEmit":true,"target":"ES2022","module":"ESNext","moduleResolution":"Bundler","lib":["ES2022","DOM"],"skipLibCheck":true,"strict":false},"files":["SAND.js"]}' > tsconfig.json
npx tsc -p tsconfig.json --pretty false

echo "===== Structured fallback and pure core gates ====="
set -euo pipefail
cp SAND.js PRIME_TEST.mjs
printf '\nexport { safeMathEvaluate, zonedLocalToUtc, nextScheduleUtc, isPrivateHost, validateRecipeProposal, extractJsonObject, callAiChain };\n' >> PRIME_TEST.mjs
node --input-type=module <<'NODE'
import {safeMathEvaluate,zonedLocalToUtc,nextScheduleUtc,isPrivateHost,validateRecipeProposal,extractJsonObject,callAiChain} from './PRIME_TEST.mjs';
const ok=(x,m)=>{if(!x)throw Error(m)};
ok(safeMathEvaluate('(10+5)*2^2')===60,'calculator');
ok(safeMathEvaluate('2^3^2')===512,'right associative exponent');
ok(zonedLocalToUtc('2026-08-19T19:00','Africa/Cairo')==='2026-08-19T16:00:00.000Z','Cairo DST');
ok(nextScheduleUtc({type:'interval',minutes:15,timezone:'Africa/Cairo'},'2026-08-18T18:00:00.000Z')==='2026-08-18T18:15:00.000Z','interval schedule');
ok(isPrivateHost('127.0.0.1')&&isPrivateHost('192.168.1.1')&&!isPrivateHost('api.example.com'),'SSRF guard');
let rejected=false;try{extractJsonObject('not json')}catch(e){rejected=e.code==='AI_BAD_JSON'}ok(rejected,'non-JSON rejection');
let blocked=false;try{validateRecipeProposal({ALLOW_FORGED_PUBLIC_GET:'true'},{name:'ext.bad',method:'GET',url_template:'https://127.0.0.1/x',input_schema:{}})}catch(e){blocked=e.code==='FORGE_SSRF_BLOCKED'}ok(blocked,'forge SSRF guard');
const calls=[];
const env={OMNIAI_API_KEY:'x',DB:{prepare(){return{bind(...a){return{run:async()=>{calls.push(a);return{success:true}}}}}}},OMNIAI_SERVICE:{fetch:async(req)=>{const b=await req.json();let content='{"ok":true}';if(b.model==='groq::openai/gpt-oss-120b')content='bad prose';else if(b.model==='groq::qwen/qwen3.6-27b')content='```json\n{"ok":true}\n```';return new Response(JSON.stringify({choices:[{message:{content}}]}),{status:200,headers:{'content-type':'application/json'}})}}};
const r=await callAiChain(env,{chatId:'1',updateId:1,messages:[{role:'user',content:'x'}],validate:t=>{const j=extractJsonObject(t);if(j.ok!==true)throw Error('bad');return j}});
ok(r.modelId==='groq::qwen/qwen3.6-27b','structured output must fall through bad primary');
ok(r.parsed.ok===true&&calls.length===2,'structured fallback result/logging');
console.log('PURE_CORE_GATES_OK');
NODE

echo "===== Probe fixed three model routes ====="
set -euo pipefail
test -n "$OKEY"
node - <<'NODE'
const key=process.env.OKEY, ep='https://omniai-engine.ahmeddoba91.workers.dev/v1/chat/completions';
const models=['groq::openai/gpt-oss-120b','groq::qwen/qwen3.6-27b','gemini::gemini-3.5-flash-lite'];
(async()=>{for(const model of models){const c=new AbortController(),tm=setTimeout(()=>c.abort(),12000),t=Date.now();try{const r=await fetch(ep,{method:'POST',headers:{Authorization:'Bearer '+key,'Content-Type':'application/json'},body:JSON.stringify({model,messages:[{role:'user',content:'رد بكلمة تمام فقط'}],temperature:0,max_tokens:50,stream:false}),signal:c.signal});const d=await r.json();const text=d?.choices?.[0]?.message?.content||d?.output_text||d?.text||'';console.log(JSON.stringify({model,status:r.status,ms:Date.now()-t,nonempty:Boolean(String(text).trim())}));if(!r.ok||!String(text).trim())throw Error('route failed '+model+' '+r.status)}finally{clearTimeout(tm)}}})().catch(e=>{console.error(e);process.exit(1)});
NODE

echo "===== Deploy isolated PRIME staging ====="
set -euo pipefail
test -n "$BOT"; test -n "$OKEY"; test -n "$CHAT"
cat > "$CFG" <<'EOF'
{"name":"sand-prime-staging","main":"SAND.js","compatibility_date":"2026-08-18","vars":{"PUBLIC_BOT":"false","ALLOW_FORGED_PUBLIC_GET":"true"},"d1_databases":[{"binding":"DB","database_name":"superagent-v106-staging","database_id":"65df37bc-973d-4bfe-a3f4-7c049e570697"}],"services":[{"binding":"OMNIAI_SERVICE","service":"omniai-engine"}]}
EOF
npx wrangler deploy --config "$CFG" >/dev/null
W=$(openssl rand -hex 32); K=$(openssl rand -hex 32)
echo "::add-mask::$W"; echo "::add-mask::$K"
export W K
for p in "TELEGRAM_BOT_TOKEN:$BOT" "OMNIAI_API_KEY:$OKEY" "TELEGRAM_WEBHOOK_SECRET:$W" "SETUP_KEY:$K" "ALLOWED_CHAT_ID:$CHAT"; do n=${p%%:*}; v=${p#*:}; printf '%s' "$v" | npx wrangler secret put "$n" --config "$CFG" >/dev/null; done

echo "===== Health, secret propagation and selftest ====="
set -euo pipefail
for i in {1..25}; do
  code=$(curl -sS -o /tmp/h -w '%{http_code}' "$URL/health" || true)
  if [ "$code" = 200 ] && node -e 'const x=JSON.parse(require("fs").readFileSync("/tmp/h"));process.exit(x.ok&&x.version==="1.0.0"&&x.schema_version==="3"&&x.prime_schema_version==="1"&&x.semantic_routing===true&&x.keyword_routing===false?0:1)'; then break; fi
  sleep 2
done
cat /tmp/h
node -e 'const x=JSON.parse(require("fs").readFileSync("/tmp/h"));if(!x.ok||x.version!=="1.0.0"||x.schema_version!=="3"||x.prime_schema_version!=="1"||!x.bindings?.OMNIAI_SERVICE||x.keyword_routing!==false)throw Error(JSON.stringify(x))'
code=0
for i in {1..25}; do code=$(curl -sS -o /tmp/auth -w '%{http_code}' -X POST -H 'content-type: application/json' -H "X-Telegram-Bot-Api-Secret-Token: $W" -d '{}' "$URL/telegram" || true); echo "secret probe $i http=$code"; [ "$code" = 200 ] && break; sleep 2; done
test "$code" = 200
curl -fsS -H "x-sand-admin: $K" "$URL/admin/selftest" >/tmp/selftest.json
cat /tmp/selftest.json
node -e 'const x=JSON.parse(require("fs").readFileSync("/tmp/selftest.json"));if(!x.ok||!x.tests?.every(t=>t.ok))throw Error(JSON.stringify(x))'

echo "===== Clean staging chat state ====="
set -euo pipefail
SQL="DELETE FROM sand_prime_operation_steps WHERE operation_id IN (SELECT id FROM sand_prime_operations WHERE chat_id='$CHAT');DELETE FROM sand_prime_list_items WHERE chat_id='$CHAT';DELETE FROM sand_prime_confirmations WHERE chat_id='$CHAT';DELETE FROM sand_prime_audit WHERE chat_id='$CHAT';DELETE FROM sand_prime_outbox WHERE chat_id='$CHAT';DELETE FROM sand_prime_forge_requests WHERE chat_id='$CHAT';DELETE FROM sand_prime_automations WHERE chat_id='$CHAT';DELETE FROM sand_prime_contacts WHERE chat_id='$CHAT';DELETE FROM sand_prime_calendar_events WHERE chat_id='$CHAT';DELETE FROM sand_prime_notes WHERE chat_id='$CHAT';DELETE FROM sand_prime_lists WHERE chat_id='$CHAT';DELETE FROM sand_prime_tasks WHERE chat_id='$CHAT';DELETE FROM sand_prime_reminders WHERE chat_id='$CHAT';DELETE FROM sand_prime_chat_state WHERE chat_id='$CHAT';DELETE FROM sand_prime_entities WHERE chat_id='$CHAT';DELETE FROM sand_prime_memories WHERE chat_id='$CHAT';DELETE FROM sand_prime_operations WHERE chat_id='$CHAT';DELETE FROM sand_prime_decisions WHERE chat_id='$CHAT';DELETE FROM sand_ai_calls WHERE chat_id='$CHAT';DELETE FROM sand_ai_messages WHERE chat_id='$CHAT';DELETE FROM sand_core_chat_leases WHERE chat_id='$CHAT';DELETE FROM sand_core_ch
